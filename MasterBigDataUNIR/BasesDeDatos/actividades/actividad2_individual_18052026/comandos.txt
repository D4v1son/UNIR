db.books.aggregate([
  { $match: { pageCount: { $gt: 0 } } },
  {
    $group: {
      _id: null,
      max_paginas: { $max: "$pageCount" },
      min_paginas: { $min: "$pageCount" },
      media_paginas: { $avg: "$pageCount" }
    }
  }
])

# f) Elige una variable categórica y calcula el total de documentos agrupando por dicha variable.
